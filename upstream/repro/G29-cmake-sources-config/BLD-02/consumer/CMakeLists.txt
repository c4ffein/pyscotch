cmake_minimum_required(VERSION 3.10)
project(consumer LANGUAGES C)
find_package(SCOTCH REQUIRED)
message(STATUS "SCOTCH_FOUND=${SCOTCH_FOUND} SCOTCH_LIBESMUMPS_FOUND=${SCOTCH_LIBESMUMPS_FOUND}")
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE SCOTCH::scotch SCOTCH::scotcherr)
if(NOT "${SUFFIX}" STREQUAL "")
  target_compile_definitions(consumer PRIVATE SUFFIX=${SUFFIX})
endif()
get_target_property(defs SCOTCH::scotch INTERFACE_COMPILE_DEFINITIONS)
message(STATUS "SCOTCH::scotch INTERFACE_COMPILE_DEFINITIONS=[${defs}]")
